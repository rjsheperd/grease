** DONE 8.1 Document ~@objc~ bridge pattern

Written as ~examples/swift/swift_bridge.clj~. Key findings:

- Swift classes register as ~ModuleName.ClassName~ (e.g. ~MobileTest.AppDelegate~)
- Use ~@objc(ClassName)~ annotation to override to a bare name accessible as
  ~(get-class "ClassName")~ instead of ~(get-class "MobileTest.ClassName")~
- ~@objc var message: String~ synthesises both ~"message"~ getter and
  ~"setMessage:"~ setter — do NOT also define ~func setMessage~, that conflicts
- Static properties (~@objc static let shared~) are class methods on the
  metaclass, callable via ~(msg-send :pointer cls "shared")~

#+begin_src clojure
;; Access a Swift singleton from Clojure:
(def hook (grease.ios.objc/msg-send :pointer (get-class "GreaseHook") "shared"))

;; Read an @objc property:
(f/nsstring->str (grease.ios.objc/msg-send :pointer hook "message"))
;; => "No message yet"

;; Write an @objc property (via synthesized setter):
(grease.ios.objc/msg-send :void hook "setMessage:" :pointer (f/->nsstring "Hello!"))
#+end_src

@objc func fetchURL(_ urlString: String) {
    message = "Fetching \(urlString)..."
    Task {
        let (data, resp) = try await URLSession.shared.data(from: url)
        let status = (resp as? HTTPURLResponse)?.statusCode ?? 0
        message = "HTTP \(status): \(String(data.prefix(200)))"

}
